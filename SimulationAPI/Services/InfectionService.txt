public class Quadtree
{
    private const int MAX_AGENTS = 10; // max agents before subdividing
    private readonly double MinX, MaxX, MinY, MaxY;
    private readonly List<Agent> Agents = new();
    private Quadtree[]? Children = null;

    public Quadtree(double minX, double maxX, double minY, double maxY)
    {
        MinX = minX;
        MaxX = maxX;
        MinY = minY;
        MaxY = maxY;
    }

    public void Insert(Agent agent)
    {
        if (Children != null)
        {
            int quad = GetQuadrant(agent);
            Children[quad].Insert(agent);
            return;
        }

        Agents.Add(agent);

        if (Agents.Count > MAX_AGENTS)
            Subdivide();
    }

    private void Subdivide()
    {
        double midX = (MinX + MaxX) / 2;
        double midY = (MinY + MaxY) / 2;

        Children = new Quadtree[4];
        Children[0] = new Quadtree(MinX, midX, MinY, midY); // bottom-left
        Children[1] = new Quadtree(midX, MaxX, MinY, midY); // bottom-right
        Children[2] = new Quadtree(MinX, midX, midY, MaxY); // top-left
        Children[3] = new Quadtree(midX, MaxX, midY, MaxY); // top-right

        // re-insert existing agents into children
        foreach (var a in Agents)
        {
            int quad = GetQuadrant(a);
            Children[quad].Insert(a);
        }
        Agents.Clear();
    }

    private int GetQuadrant(Agent agent)
    {
        double midX = (MinX + MaxX) / 2;
        double midY = (MinY + MaxY) / 2;

        bool left = agent.CurrentLon < midX;
        bool bottom = agent.CurrentLat < midY;

        if (left && bottom) return 0;
        if (!left && bottom) return 1;
        if (left && !bottom) return 2;
        return 3;
    }

    public void Query(double lat, double lon, double radius, List<Agent> result)
    {
        if (!IntersectsCircle(lat, lon, radius)) return;

        if (Children != null)
        {
            foreach (var child in Children)
                child.Query(lat, lon, radius, result);
        }
        else
        {
            double radiusSq = radius * radius;
            foreach (var agent in Agents)
            {
                double dx = agent.CurrentLat - lat;
                double dy = agent.CurrentLon - lon;
                if ((dx * dx + dy * dy) <= radiusSq)
                    result.Add(agent);
            }
        }
    }

    private bool IntersectsCircle(double lat, double lon, double radius)
    {
        double closestX = Math.Max(MinX, Math.Min(lon, MaxX));
        double closestY = Math.Max(MinY, Math.Min(lat, MaxY));
        double dx = closestX - lon;
        double dy = closestY - lat;
        return (dx * dx + dy * dy) <= radius * radius;
    }
}









public static class DiseaseSpread
{
    private static readonly Random Rnd = new();

    public static void Spread(List<Agent> agents, double infectionRadius)
    {
        // 1. Build quadtree
        double minLat = agents.Min(a => a.CurrentLat);
        double maxLat = agents.Max(a => a.CurrentLat);
        double minLon = agents.Min(a => a.CurrentLon);
        double maxLon = agents.Max(a => a.CurrentLon);

        var qt = new Quadtree(minLon, maxLon, minLat, maxLat);
        foreach (var agent in agents)
            qt.Insert(agent);

        // 2. For each agent, find nearby agents
        foreach (var agent in agents)
        {
            var nearby = new List<Agent>();
            qt.Query(agent.CurrentLat, agent.CurrentLon, infectionRadius, nearby);

            foreach (var other in nearby)
            {
                if (agent.Id == other.Id) continue;
                TryInfect(agent, other);
            }
        }
    }

    private static void TryInfect(Agent a, Agent b)
    {
        if (a.IsInfected == b.IsInfected) return;

        double baseChance = a.Disease.CalculateInfectionChance(a, b);

        // Boost infection indoors
        if (a.CurrentBuilding != null && a.CurrentBuilding == b.CurrentBuilding)
        {
            baseChance *= 3.0; // indoor factor
        }

        if (Rnd.NextDouble() < baseChance)
        {
            if (!a.IsInfected) a.IsInfected = true;
            else b.IsInfected = true;
        }
    }
}




public void SimulationTick(List<Agent> agents, double infectionRadius)
{
    // 1. Move agents based on routine (update CurrentLat/CurrentLon/CurrentBuilding)
    UpdateAgentPositions(agents);

    // 2. Spread disease
    DiseaseSpread.Spread(agents, infectionRadius);

    // 3. Optionally snapshot positions every N ticks
    SaveSnapshotIfNeeded(agents);
}
